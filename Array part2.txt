// TWO sum

->indices two pointer approach
 import java.util.*;

class Solution {
    public int[] twoSum(int[] nums, int target) {

        int n = nums.length;

        // pair array (value, index)
        int[][] arr = new int[n][2];

        for (int i = 0; i < n; i++) {
            arr[i][0] = nums[i];
            arr[i][1] = i;
        }

        // sort by value
        Arrays.sort(arr, (a, b) -> a[0] - b[0]);

        int left = 0, right = n - 1;

        while (left < right) {

            int sum = arr[left][0] + arr[right][0];

            if (sum == target) {
                return new int[]{arr[left][1], arr[right][1]};
            }
            else if (sum < target) {
                left++;
            }
            else {
                right--;
            }
        }

        return new int[]{-1, -1};
    }
}




// Sort Array of 0 1 and 2

class Solution {
    public void sortColors(int[] nums) {
        int n = nums.length;

        int c=0,c1=0,c2=0;

        for(int i=0;i<n;i++)

        {

            if (nums[i]==0)
            c++;
            else if (nums[i]==1)
            c1++;
            else 
            c2++;
        }

        for(int i=0;i<c;i++)
        {
            nums[i]=0;
        }
        for(int i=c;i<c+c1;i++)
        {
            nums[i]=1;
        }
        for(int i=c+c1;i<n;i++)
        {
            nums[i]=2;
        }
    }
}
